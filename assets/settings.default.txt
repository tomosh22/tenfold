# tenfold default settings (embedded in the binary; edit a copy, not this file).
#
# Format: one `key = value` per line, dotted keys (no [sections]). `#` starts a comment only as the first
# non-blank character of a line (colours contain '#'). Types: bool (true/false), int, float, string (optional
# "quotes"), colour (#rrggbb or #rrggbbaa), font ("family or path[, size in pt]"; empty = unset).
# Later layers override earlier ones: this file, <user config dir>/settings.txt, <workspace>/.tenfold/settings.txt.
# Undo grouping (undoGrouping, undoGroupPauseMs) lives in <workspace>/.tenfold/workspace.json, not here.

# OPEN_QUESTIONS #4 (answered): soft word wrap on, smooth scrolling off.
editor.smoothScroll = false
editor.wordWrap = true
editor.caretBlink = true
editor.caretBlinkMs = 530
editor.highlightCurrentLine = true
editor.wheelLines = 3
editor.tabWidth = 4
# 0 = off (10x VerticalRulers is empty by default; tenfold draws one guide).
editor.columnGuide = 0
editor.showWhitespace = false
# 10x ShowCollapseRegionMargin (default true there) and Visual Studio's outlining margin (on by default): a lane in the
# gutter, between the line numbers and the change strip, with a boxed [-] / [+] on every region start and a guide line
# through the open region (click to collapse / expand). A collapsed region always shows its {...} marker.
editor.showFoldGlyphs = true
# Visual Studio's C/C++ outlining options (Tools > Options > Text Editor > C/C++ > View > Outlining), all on there:
# Enable Outlining (off: no regions, no margin), Outline Pragma Regions (#pragma region / endregion), Outline Statement
# Blocks (off: braces nested in a function body are not regions; functions, classes, namespaces and comments still are).
outline.enabled = true
outline.pragmaRegions = true
outline.statementBlocks = true

# Editing conveniences (B8b: the defaults are 10x's documented defaults; the 10x setting each key mirrors is named).
# indentUnit (10x IndentMode = Tabs): "tab" or the number of spaces of one indent level; used when detectIndent is off
# or finds nothing in the file.
editor.indentUnit = tab
# 10x AutoDetectIndentMode = false.
editor.detectIndent = false
# 10x AutoIndent: Enter copies the previous line's indentation (one level deeper after { ( [ or an operator).
editor.autoIndent = true
# 10x AutoIndentBraces: '}' typed as the first non-blank character dedents to the line of its '{'.
editor.autoIndentBraces = true
# 10x FormatBracesOnEnterSplit: Enter between { and } moves the closer to its own line.
editor.formatBracesOnEnterSplit = false
# 10x AutoAddCloseBrace: typing { inserts }; typing the closer steps over it, Backspace removes the empty pair.
editor.autoAddCloseBrace = false
# 10x AutoAddCloseBracket: the same for ( and [.
editor.autoAddCloseBracket = false
# tenfold only (10x documents no setting for quotes): " and ' insert a second quote and wrap a selection.
editor.autoCloseQuotes = false
# 10x BracketSelectedText: typing ( [ { over a selection encloses it.
editor.bracketSelectedText = true
# Not documented by 10x; stays off.
editor.trimTrailingWhitespace = false
# 10x CommentBlockUsesLineComments: ToggleComment with a selection uses // line comments (false: /* */ around the selection).
editor.commentBlockUsesLineComments = true

# Unsaved changes and files changed on disk (OPEN_QUESTIONS #5). confirmUnsaved matches 10x (RestoreUnsavedFiles =
# false: it prompts); 10x documents nothing about files changed on disk, that default is a guess.
# confirmUnsaved: closing the last tab of a modified file, closing the window and App.Quit ask Save / Don't Save / Cancel.
editor.confirmUnsaved = true
# reloadExternalChanges: auto (reload a file that changed on disk unless it has unsaved edits, then ask), prompt (always
# ask) or never (leave the buffer alone, only say so in the status bar).
editor.reloadExternalChanges = auto

# New files, search and zoom (10x DefaultLineEnding, AutoNameUntitledFiles, InternetSearchCommand).
# defaultLineEnding: line ending of a new untitled file, CRLF or LF (10x's default is CRLF; any other text is CRLF).
editor.defaultLineEnding = CRLF
# Line endings follow Visual Studio: every line keeps its own ending, Enter copies the ending of the line it is
# typed on (replicateNewLine; false = always defaultLineEnding), pasted text keeps its own line breaks
# (pasteKeepsLineBreaks; false = they become the ending Enter would use), and a file with mixed endings asks
# "Do you want to normalize the line endings?" when it is opened (checkLineEndings).
editor.replicateNewLine = true
editor.pasteKeepsLineBreaks = true
editor.checkLineEndings = true
# autoNameUntitledFiles: an untitled file takes its name from the first word typed in it (the name is only a
# suggestion for the Save As prompt until the file is saved).
editor.autoNameUntitledFiles = false
# internetSearchCommand: InternetSearch opens this address with the word under the caret (or the selection) appended,
# percent-encoded, in the default browser.
editor.internetSearchCommand = https://www.google.com/search?q=
# pasteHistorySize: how many recent clipboard texts PasteHistory (Control Shift V) remembers (10x documents no size;
# tenfold's default is 20). 0 turns the history off.
editor.pasteHistorySize = 20

# Unset: the app discovers a monospace font, trying font.fallbacks in order (OPEN_QUESTIONS #7).
font.editor =
font.fallbacks = Cascadia Mono, Consolas
font.size = 10

# Theme name: workspace .tenfold/themes/<name>.txt, then user themes/<name>.txt, then built-in (dark, light, vs-dark).
ui.theme = vs-dark

# Show "<- 3 | 1 ->" back/forward counts in the status bar.
nav.statusCounts = false

# Windows: also read the user's %APPDATA%\10x\Settings\KeyMappings.10x_settings as a key-binding layer.
keys.import10x = true

# Symbol search shows a preview of the target line on the right (OPEN_QUESTIONS #6: not documented by 10x).
picker.preview = true

# Workspace explorer (W8f): keep the active file selected in the tree while the panel is open (10x:
# TrackActiveFileInWorkspaceTree).
explorer.trackActiveFile = false

# Version control (C3c). vcs.diffTool: the external diff program of Vcs.DiffFileExternal; empty = auto: on Windows
# p4merge.exe from PATH, then %ProgramFiles%\Perforce\p4merge.exe; elsewhere p4merge from PATH.
# vcs.diffToolArgs: its arguments, split like a command line (double quotes group). Placeholders: {base} (temp file
# with the committed text), {current} (temp file with the buffer), {baseTitle}, {currentTitle}, {file} (the real
# path). A value wrapped in quotes loses the outer pair, hence the doubled quotes below. For p4merge with these
# default arguments the titles are passed as well (-nl "<file>#have" -nr "<file> (unsaved)").
vcs.diffTool =
vcs.diffToolArgs = ""{base}" "{current}""
# vcs.p4vPath: the p4v program PerforceShowInP4v starts (empty = `p4v` from PATH).
vcs.p4vPath =

# Terminal panel (W8f). The first five mirror 10x's TerminalCommand (cmd.exe), ClaudeTerminalCommand (claude),
# CursorTerminalCommand (cmd /c agent), TerminalFont and TerminalFontSize; command lines are split like a shell command
# line (double quotes group). Off Windows, 10x's Windows defaults are translated: `cmd.exe` means $SHELL (else /bin/sh)
# and a leading `cmd /c` is dropped. terminal.font empty = the editor font; terminal.fontSize 0 = the editor font size.
terminal.command =
terminal.claudeCommand = claude
terminal.cursorCommand = cmd /c agent
terminal.font =
terminal.fontSize = 0
# tenfold only (10x documents neither): lines of scrollback per terminal, and whether a terminal tab closes by itself
# when its program exits (false: the tab keeps the final screen and "[process exited with code N]").
terminal.scrollbackLines = 10000
terminal.closeOnExit = false
# tenfold only, Visual Studio's terminal behaviour (docs/VS_BEHAVIOUR.md sec. 12): Ctrl+C always sends ^C to the program
# (copy is Ctrl+Shift+C / Ctrl+Insert, paste Ctrl+Shift+V / Shift+Insert / right click). true: Ctrl+C copies (and
# clears) the selection when there is one, like Windows Terminal; with no selection it is still ^C.
terminal.ctrlCCopiesSelection = false
