# tenfold default settings (embedded in the binary; edit a copy, not this file).
#
# Format: one `key = value` per line, dotted keys (no [sections]). `#` starts a comment only as the first
# non-blank character of a line (colours contain '#'). Types: bool (true/false), int, float, string (optional
# "quotes"), colour (#rrggbb or #rrggbbaa), font ("family or path[, size in pt]"; empty = unset).
# Later layers override earlier ones: this file, <user config dir>/settings.txt, <workspace>/.tenfold/settings.txt.
# Undo grouping (undoGrouping, undoGroupPauseMs) lives in <workspace>/.tenfold/workspace.json, not here.

# OPEN_QUESTIONS #4 (answered): soft word wrap on, smooth scrolling off.
editor.smoothScroll = false
editor.wordWrap = true
editor.caretBlink = true
editor.caretBlinkMs = 530
editor.highlightCurrentLine = true
editor.wheelLines = 3
editor.tabWidth = 4
# 0 = off (10x VerticalRulers is empty by default; tenfold draws one guide).
editor.columnGuide = 0
editor.showWhitespace = false

# Editing conveniences (B8b: the defaults are 10x's documented defaults; the 10x setting each key mirrors is named).
# indentUnit (10x IndentMode = Tabs): "tab" or the number of spaces of one indent level; used when detectIndent is off
# or finds nothing in the file.
editor.indentUnit = tab
# 10x AutoDetectIndentMode = false.
editor.detectIndent = false
# 10x AutoIndent: Enter copies the previous line's indentation (one level deeper after { ( [ or an operator).
editor.autoIndent = true
# 10x AutoIndentBraces: '}' typed as the first non-blank character dedents to the line of its '{'.
editor.autoIndentBraces = true
# 10x FormatBracesOnEnterSplit: Enter between { and } moves the closer to its own line.
editor.formatBracesOnEnterSplit = false
# 10x AutoAddCloseBrace: typing { inserts }; typing the closer steps over it, Backspace removes the empty pair.
editor.autoAddCloseBrace = false
# 10x AutoAddCloseBracket: the same for ( and [.
editor.autoAddCloseBracket = false
# tenfold only (10x documents no setting for quotes): " and ' insert a second quote and wrap a selection.
editor.autoCloseQuotes = false
# 10x BracketSelectedText: typing ( [ { over a selection encloses it.
editor.bracketSelectedText = true
# Not documented by 10x; stays off.
editor.trimTrailingWhitespace = false

# Unsaved changes and files changed on disk (OPEN_QUESTIONS #5). confirmUnsaved matches 10x (RestoreUnsavedFiles =
# false: it prompts); 10x documents nothing about files changed on disk, that default is a guess.
# confirmUnsaved: closing the last tab of a modified file, closing the window and App.Quit ask Save / Don't Save / Cancel.
editor.confirmUnsaved = true
# reloadExternalChanges: auto (reload a file that changed on disk unless it has unsaved edits, then ask), prompt (always
# ask) or never (leave the buffer alone, only say so in the status bar).
editor.reloadExternalChanges = auto

# Unset: the app discovers a monospace font, trying font.fallbacks in order (OPEN_QUESTIONS #7).
font.editor =
font.fallbacks = Cascadia Mono, Consolas
font.size = 10

# Theme name: workspace .tenfold/themes/<name>.txt, then user themes/<name>.txt, then built-in (dark, light, vs-dark).
ui.theme = vs-dark

# Show "<- 3 | 1 ->" back/forward counts in the status bar.
nav.statusCounts = false

# Windows: also read the user's %APPDATA%\10x\Settings\KeyMappings.10x_settings as a key-binding layer.
keys.import10x = true

# Symbol search shows a preview of the target line on the right (OPEN_QUESTIONS #6: not documented by 10x).
picker.preview = true

# Version control (C3c). vcs.diffTool: the external diff program of Vcs.DiffFileExternal; empty = auto: on Windows
# p4merge.exe from PATH, then %ProgramFiles%\Perforce\p4merge.exe; elsewhere p4merge from PATH.
# vcs.diffToolArgs: its arguments, split like a command line (double quotes group). Placeholders: {base} (temp file
# with the committed text), {current} (temp file with the buffer), {baseTitle}, {currentTitle}, {file} (the real
# path). A value wrapped in quotes loses the outer pair, hence the doubled quotes below. For p4merge with these
# default arguments the titles are passed as well (-nl "<file>#have" -nr "<file> (unsaved)").
vcs.diffTool =
vcs.diffToolArgs = ""{base}" "{current}""
