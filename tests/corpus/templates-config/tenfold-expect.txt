fast def use_run_int -> Policy_safe_run

def use_cost_int -> cost_small

def use2_cost -> cost_small
xfail decl use_cost_dbl -> cost_big  # clangd: cost<Small T>() and cost() requires(!Small<T>) share one USR (constraints are not part of it), the index's canonical declaration is the first
xfail refs cost_small = cost_small use_cost_int use2_cost  # same: the references of cost_big are reported too
refs cost_small >= cost_small use_cost_int use2_cost
refs cost_big >= cost_big use_cost_dbl

hover use_cost_dbl ~ "cost"
rename Policy_safe_run exec
  Policy_safe_run use_run_int use_run_dbl use2_run_int use_run_in_twice
