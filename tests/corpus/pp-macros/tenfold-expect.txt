# pp-macros (W9c): object-like / function-like macros, declare-by-macro, token pasting, stringising, variadic macros,
# __VA_OPT__, #define/#undef order, #if chains. Standard-library free; compile_commands.json (clang++ -std=c++20).
# Where a macro generates a declaration, the correct landing spot is the place the name is spelled in the source (the
# macro argument) or the macro invocation that produced it; both are accepted where a name is pasted.

# ---- fast tier ----
fast def use_helper_sum -> helper_sum
fast def use_max_items -> MAX_ITEMS
fast def use_square -> SQUARE
xfail fast def use_Gadget -> Gadget  # names declared through macro arguments (DECLARE_CLASS(Gadget)) are not in the lexer index
xfail fast def use_compute_b -> compute_b  # same: MAKE_FUNC(compute_b, ...)
xfail fast def use_warm -> Hue_Warm  # same: DECLARE_ENUM(Hue, Warm, Cold)
fast def use_item_id -> Item_id
fast def use_twice_arg -> twice_fn
fast sym "helper_sum" -> helper_sum
xfail fast sym "Gadget" ∋ Gadget  # same: not declared as a class in the lexer index
# the lexer index does not know that this spelling is the function and not the macro
fast def use_clamp01_paren -> clamp01_fn
# #undef / #define order is invisible to the lexer index
xfail fast def use_level2 -> LEVEL_second  # fast index cannot order redefinitions
fast def use_late_before -> var_late

# ---- a function-like macro and a function with the same name ----
def use_clamp01_macro -> clamp01
xfail def use_clamp01_paren -> clamp01_fn  # clangd treats a function-like macro name not followed by "(" as a macro use
xfail def use_clamp01_ptr -> clamp01_fn  # same
hover use_clamp01_macro ~ "#define clamp01(v)"
xfail hover use_clamp01_paren ~ "clamp01" "int"  # same: clangd hovers the macro
xfail refs clamp01_fn = clamp01_fn use_clamp01_paren use_clamp01_ptr  # same: the uses of the function are reported as the macro's
refs clamp01 = clamp01 use_clamp01_macro
