# ---- overloads ordered by constraints ----
def use_describe_num -> describe_num
def use_describe_sized -> describe_sized
def use_describe_sized2 -> describe_sized
def use_describe_any -> describe_any
# inside a template the argument is dependent: any candidate is acceptable
def use_describe_in_tmpl -> describe_num | describe_cont | describe_sized | describe_any
def use_dispatch -> dispatch
def use_same_bytes -> same_bytes
refs describe_sized >= describe_sized use_describe_sized use_describe_sized2
refs describe_any >= describe_any use_describe_any
refs describe_num >= describe_num use_describe_num
# clang's USR ignores type-constraints, so describe(const Numeric auto&), (const Container auto&), (const Sized auto&)
# and (const auto&) are ONE symbol in clangd's index: references and rename of one overload hit all call sites.
xfail refs describe_num = describe_num use_describe_num  # clangd: the four describe() overloads share one USR (constraints are not part of it)
xfail refs describe_sized = describe_sized use_describe_sized use_describe_sized2  # same
xfail rename describe_num describeNumber  # same: also renames the calls that resolve to the other three overloads
  describe_num use_describe_num
