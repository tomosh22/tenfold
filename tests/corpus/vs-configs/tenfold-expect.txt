# vs-configs (W9c): one solution, two projects (core, app), four solution configurations:
#   Debug|x64, Release|x64, Debug|x86 (solution x86 maps to the projects' Win32), Profile|x64 (custom configuration;
#   app has no Profile configuration and is mapped to Release|x64 by the solution).
# Everything is standard-library free on purpose (same meaning on every platform; vs-std covers the standard library in .sln mode).
# Per-configuration: PreprocessorDefinitions, ForcedIncludeFiles (Debug), AdditionalIncludeDirectories built from
# $(Configuration) and $(Platform), ExcludedFromBuild conditions, implicit _DEBUG / _WIN64 defines, shared headers.

# ---- fast tier, before clangd has parsed anything (the lexer index does not know configurations) ----

fast def use_app_mode -> app_mode_release | app_mode_debug

fast sym "x64_only_fn" ∋ x64_only_fn
fast sym "dbg_helper" ∋ dbg_helper_def
fast sym "dbg_only_free" -> dbg_only_free
fast sym "app_mode" ∋ app_mode_release app_mode_debug
fast def use_core_level -> level_dbg
fast def use_log_app -> log_plain
