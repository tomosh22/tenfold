# CRTP. Judgement calls are marked "(judgement)": the C++ rules do not say which declaration an IDE must show.

# ---- virtual functions of a class template ----
def use_task_run -> Task_run
def use_task_name -> Task_name
def use_task_run_f -> Task_run
# Task<int>::run is overridden by IntTask (and BigTask); Task<float>::run only by FloatTask. clangd answers nothing
# at all for the virtual members of a class template: the overrides of an instantiation are not indexed as relations.
xfail impl Task_run >= IntTask_run FloatTask_run  # clangd: no implementations for virtual members of class templates
xfail impl use_task_run = IntTask_run BigTask_run  # same
xfail impl use_task_name = BigTask_name  # same
refs Task_run >= Task_run use_task_run use_task_run_f
impl IntTask_run = BigTask_run
