# pp-xmacros (W9c): X-macros. One list (a .def file or a list macro) expanded several times with different X.
# Names spelled in the list are declared once and used by every expansion; pasted names (idx_Red, op_add) are declared
# by the invocation of the per-item macro in the list.
# A .def has no compile command and is only #included; tenfold gives it the context of the translation unit that includes it
# (the includer's text before the first #include of it, with the includer's flags), so opening it no longer hides the .def's
# references from rename and find-references (docs/ARCHITECTURE.md, "Translation-unit context"). Line order no longer matters.

# ---- fast tier ----
fast def use_color_name -> color_name
fast def use_table -> color_table
fast def use_marker -> Rect_marker
fast sym "color_name" -> color_name
xfail fast def use_Red_1 -> xc_Red  # lexer index does not scan .def files; "Red" is only a macro argument there
xfail fast def use_op_add -> xo_add | xox_add  # same: op_add exists only by token pasting in ops.def's expansion
xfail fast def use_Triangle -> xs_Triangle  # same: struct generated by SHAPE(Triangle, 3) in shapes.def
xfail fast sym "Triangle" ∋ xs_Triangle  # same

# ---- colors.def: enum, switch, table, pasted index enum from one list ----
def use_Red_1 -> xc_Red
def use_Green_1 -> xc_Green
def use_Blue_1 -> xc_Blue
def use_color_name -> color_name
def use_c_arg -> c_local
def use_idx_count -> idx_count
def use_idx_Blue -> xcx_Blue | xc_Blue
def use_table -> color_table
def use_bits -> ColorInfo_bits
def use_Color_c -> Color
def use_Color_param -> Color
def use_Color_in_info -> Color
def Color_in_case -> Color
def Color_in_table -> Color
typedef use_c_arg -> Color
typedef use_table -> ColorInfo
hover use_Red_1 ~ "Red" "= 1"
hover use_Blue_1 ~ "Blue" "= 4"
hover use_color_name ~ "color_name" "const char"
hover use_idx_count ~ "idx_count"
# every use is spelled in the .def, the case/table expansions refer to the same token: asked from a use site
#? xfail refs use_Red_1 = xc_Red use_Red_1   # DISABLED: clangd answers either way between runs (XFAIL in some, XPASS in others) for an enumerator whose other spellings are macro-argument tokens of an included .def; not an index-timing or TU-context effect
refs Color >= Color use_Color_c use_Color_param use_Color_in_info
refs color_name = color_name use_color_name
refs ColorInfo_bits = ColorInfo_bits use_bits
refs idx_count = idx_count use_idx_count

# ---- ops.def: functions generated by pasting op_ with the list's name ----
def use_op_add -> xo_add | xox_add
def use_op_sub -> xo_sub | xox_sub
def use_op_mul -> xo_mul | xox_mul
def use_fn_member -> OpEntry_fn
hover use_op_add ~ "op_add" "int"
hover use_fn_member ~ "fn" "OpFn"
refs use_op_add >= use_op_add
refs OpEntry_fn = OpEntry_fn use_fn_member
refs op_table = op_table use_op_table

# ---- fields.h: list macro with the per-item macro passed as an argument ----
def use_width -> fl_width | fields_decl_use
def use_height -> fl_height | fields_decl_use
def use_ratio -> fl_ratio | fields_decl_use
def use_marker -> Rect_marker
def use_count -> Rect_count
def use_DECL_FIELD -> DECL_FIELD
def fields_decl_use -> FIELDS
def fields_init_use -> FIELDS
def fields_count_use -> FIELDS
hover use_width ~ "width" "int"
hover use_ratio ~ "ratio" "double"
refs FIELDS = FIELDS fields_decl_use fields_init_use fields_count_use
xfail refs DECL_FIELD >= DECL_FIELD use_DECL_FIELD  # clangd: a macro name passed as an argument and expanded on rescan is not a reference
refs Rect_count = Rect_count use_count
refs use_width >= use_width
refs Rect_marker = Rect_marker use_marker init_marker

# ---- shapes.def: classes and their out-of-line members from one list ----
def use_Triangle -> xs_Triangle
def use_Triangle_scope -> xs_Triangle
def use_sides -> sides_member
def use_area -> shape_area_decl | shape_area_def | xsx_Triangle
decl use_area -> shape_area_decl | shape_area_def | xsx_Triangle
hover use_Triangle ~ "Triangle" "struct"
hover use_area ~ "area" "int"
refs use_Triangle >= use_Triangle

# ---- workspace symbols on a clean session ----
sym "Color" -> Color
sym "color_name" -> color_name
sym "Red" ∋ xc_Red
sym "idx_Blue" ∋ xcx_Blue
sym "color_table" -> color_table
sym "op_mul" ∋ xox_mul
sym "op_table" -> op_table
sym "OpEntry" -> OpEntry
sym "Rect" -> Rect
sym "FIELDS" ∋ FIELDS
sym "Triangle" -> xs_Triangle
sym "Hexagon" ∋ xs_Hexagon

# ---- rename on a clean session ----
xfail rename use_Red_1 Crimson  # clangd renames the use but not the token Red in the X(Red, ...) line of colors.def (a macro argument pasted/expanded elsewhere); the rename-safety check reports it
  xc_Red use_Red_1
rename color_name colour_label
  color_name use_color_name
rename use_Triangle Tri
  xs_Triangle use_Triangle use_Triangle_scope
rename Rect_marker flag
  Rect_marker use_marker init_marker
# the enumerator type is also named inside the #define bodies (Color::n); a rename that leaves them breaks the build
xfail rename Color Hue  # clangd misses the two Color:: spellings inside the #define X bodies of colors.cpp; the rename-safety check reports them (STRESS_RESULTS P-3)
  Color use_Color_c use_Color_param use_Color_in_info Color_in_case Color_in_table
# the member is spelled once, in the list macro; renaming it must edit that token (and so also the ctor initialiser)
xfail rename use_width w  # clangd renames the use only; the declaration token inside the FIELDS list macro is not edited
  fl_width use_width
# (rename use_idx_Blue idx_Navy and rename use_op_add plus would have to change nothing: harness cannot express that)

# ---- from here on the .def files are opened in the editor: queries located inside them ----
# A .def has no compile command of its own; queries asked from inside it are answered in the context of its includer.
xfail refs xc_Red = xc_Red use_Red_1  # clangd: .def has no TU, refs asked inside it are empty
xfail refs xc_Green = xc_Green use_Green_1  # same
refs xs_Triangle = xs_Triangle use_Triangle use_Triangle_scope
# Once the .def is open its refs/symbols stay available (it is parsed in the context of its includer, not alone).
open include/colors.def
sym "Red" ∋ xc_Red
xfail rename use_Red_1 Scarlet  # clangd: rename omits the colors.def declaration token (partial rename, silently)
  xc_Red use_Red_1
