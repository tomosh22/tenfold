  GuardBox_v use_GuardBox_v

  OnceBox_v use_OnceBox_v

  guarded_fn use_guarded_fn

  next_b use_next_b use_next_b_in_a
xfail rename use_dup_reversed dup_two  # clangd renames all three same-USR declarations and every use
  dup_value_inc2 use_dup_reversed
rename use_forced_fn forced_two
  forced_fn use_forced_fn
rename use_ForcedType ForcedKind
  ForcedType use_ForcedType
rename use_size length
  Stack_size use_size
rename use_top peek
  Stack_top Stack_top_def use_top
rename Widget_id ident
  Widget_id use_Widget_id
xfail rename use_box_double box_dbl  # clangd renames the use only; the name is spelled in "#define BOX_FN box_double", which is left, so the code stops compiling
  def_box_double use_box_double
rename comp_c third
  comp_c use_comp_c
