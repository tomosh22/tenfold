# paths-unicode: directory and file names with spaces, accents, CJK, emoji, parentheses, URI-special characters
# (% # + & = [ ] { } ' @ !) and a ~180-character relative path. Every directory holds a header
# and a source with the same stem; main.cpp includes them all. Names in the identifiers stay ASCII (see `encodings`).

# ---- sp: space ----
def sp_use -> sp_def
decl sp_use -> sp_decl
def sp_S_use -> sp_S
def sp_val_use -> sp_val
def sp_val_def -> sp_val
refs sp_val = sp_val sp_val_def sp_val_use
refs sp_decl = sp_decl sp_def sp_use
refs sp_S >= sp_S sp_S_use
hover sp_use ~ "fn_sp" "Box_sp"
hover sp_val_use ~ "val" "int"
rename sp_val value_sp
  sp_val sp_val_def sp_val_use
rename sp_decl renamed_sp
  sp_decl sp_def sp_use
sym "fn_sp" -> sp_def | sp_decl
