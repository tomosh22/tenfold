# vs-props (W9c): property sheets and MSBuild evaluation. Two projects (engine, plugin) share props\common.props, which
# imports props\$(Configuration).props if it exists (Debug.props, Release.props; Profile has none). Directory.Build.props
# adds DIRBUILD_PROPS. engine overrides <Level> after the sheets (Win32 only), and the sheets' item definitions must see
# the final property values (MSBuild evaluates ItemDefinitionGroups after all properties). Defines and include directories
# come from $(SolutionDir), $(ProjectDir), $(MSBuildThisFileDirectory), $(Configuration), $(Platform), $(SolutionName),
# $(ProjectName). Standard-library free.

# ---- fast tier ----
fast def use_level_value -> level_nine | level_three | level_one | level_none
fast def use_sheet_kind -> sheet_debug | sheet_release | sheet_none
fast sym "release_only" ∋ release_only_fn
fast sym "gen_a" ∋ gen_a_fn
fast sym "special_only" ∋ special_only_fn
fast sym "feature_impl" ∋ feature_debug feature_release
xfail fast def use_level_value -> level_three  # fast index cannot know which LEVEL_n the configuration defines (returns all four)
